return [ `${index + 1}.`, item.sleeve, item.size, custom, ฿${item.price} ].join(" | "); }).join("\n"); sheet.appendRow([ new Date(), data.orderNumber || "", data.customer?.name || "", data.customer?.phone || "", data.customer?.line || "", itemSummary, data.total || 0, data.slip?.fileName || "", slipUrl, "SLIP UPLOADED", "PENDING" ]); return json({ success:true, orderNumber: data.orderNumber, slipUrl }); } catch(error) { return json({ success:false, error:String(error) }); } } // ============================================================ // SHEET // ============================================================ function getOrderSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName( SHEET_NAME ); if(!sheet){ sheet = ss.insertSheet( SHEET_NAME ); sheet.appendRow([ "Timestamp", "Order Number", "Customer Name", "Phone", "LINE ID", "Items", "Total", "Slip File Name", "Slip URL", "Payment Status", "Order Status" ]); sheet .getRange(1,1,1,11) .setFontWeight("bold"); sheet.setFrozenRows(1); } return sheet; } // ============================================================ // DRIVE // ============================================================ function saveSlip(data) { if( !data.slip || !data.slip.base64 ){ return ""; } const folder = getDriveFolder(); const bytes = Utilities.base64Decode( data.slip.base64 ); const blob = Utilities.newBlob( bytes, data.slip.mimeType || "image/jpeg", data.slip.fileName || "payment-slip.jpg" ); const file = folder.createFile( blob ); return file.getUrl(); } function getDriveFolder() { const folders = DriveApp.getFoldersByName( DRIVE_FOLDER_NAME ); if(folders.hasNext()){ return folders.next(); } return DriveApp.createFolder( DRIVE_FOLDER_NAME ); } // ============================================================ // JSON RESPONSE // ============================================================ function json(data) { return ContentService .createTextOutput( JSON.stringify(data) ) .setMimeType( ContentService.MimeType.JSON );
